cmake_minimum_required(VERSION 3.22)
project(ConvolutionHost VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(cmake/Dependencies.cmake)
include(CTest)
add_library(ch_core STATIC src/audio/RoutingModel.cpp src/ir/IrDecoder.cpp src/audio/ConvolutionEngine.cpp src/ir/IrLoader.cpp)
target_include_directories(ch_core PUBLIC src)
target_link_libraries(ch_core PUBLIC juce::juce_dsp juce::juce_cryptography)
target_compile_definitions(ch_core PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
if(BUILD_TESTING)
  add_executable(ch_tests tests/TestMain.cpp tests/RealtimeGuard.cpp tests/RoutingTests.cpp tests/IrDecoderTests.cpp tests/ConvolutionTests.cpp tests/IrLoaderTests.cpp)
  target_link_libraries(ch_tests PRIVATE ch_core)
  add_test(NAME core COMMAND ch_tests)
endif()
