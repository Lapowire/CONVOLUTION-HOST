cmake_minimum_required(VERSION 3.22)
project(ConvolutionHost VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(cmake/PluginFormats.cmake)
include(cmake/FftBackend.cmake)
include(cmake/Dependencies.cmake)
include(CTest)
include(cmake/BuildIdentity.cmake)
if(BUILD_TESTING)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CTestCustom.cmake" "set(CTEST_CUSTOM_MAXIMUM_PASSED_TEST_OUTPUT_SIZE 65536)\n")
endif()
add_library(ch_core STATIC src/audio/BusLayout.cpp src/audio/RoutingModel.cpp src/ir/IrDecoder.cpp src/audio/ConvolutionEngine.cpp src/ir/IrLoader.cpp src/audio/ReverbProcessor.cpp src/ir/IrLibrary.cpp src/state/StateStore.cpp src/state/PresetFile.cpp)
target_include_directories(ch_core PUBLIC src)
target_link_libraries(ch_core PUBLIC juce::juce_dsp juce::juce_cryptography juce::juce_data_structures)
target_link_libraries(ch_core PUBLIC ch_fft_backend)
target_compile_definitions(ch_core PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
if(BUILD_TESTING)
  add_executable(ch_tests tests/TestMain.cpp tests/WorkerFailureTests.cpp tests/RealtimeGuard.cpp tests/BusLayoutTests.cpp tests/RoutingTests.cpp tests/IrDecoderTests.cpp tests/ConvolutionTests.cpp tests/IrLoaderTests.cpp tests/ReverbTests.cpp tests/LibraryTests.cpp tests/StateTests.cpp tests/PresetFileTests.cpp)
  target_link_libraries(ch_tests PRIVATE ch_core)
  add_test(NAME core COMMAND ch_tests)
endif()

if(CH_BUILD_PLUGIN)
  if(CH_BUILD_AAX AND CH_AAX_SDK_PATH)
    juce_set_aax_sdk_path("${CH_AAX_SDK_PATH}")
  endif()
  juce_add_plugin(ConvolutionHost
    COMPANY_NAME Lapowire
    BUNDLE_ID com.lapowire.convolutionhost
    PLUGIN_MANUFACTURER_CODE Lpwr
    PLUGIN_CODE Cvht
    PRODUCT_NAME "CONVOLUTION HOST"
    FORMATS ${CH_JUCE_FORMATS}
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(ConvolutionHost PRIVATE src/plugin/PluginProcessor.cpp src/plugin/Parameters.cpp src/ui/PluginEditor.cpp src/ui/IrDisplay.cpp src/ui/LibraryPanel.cpp src/ui/RoutingPanel.cpp src/ui/MatrixEditor.cpp)
  target_link_libraries(ConvolutionHost PRIVATE ch_core juce::juce_audio_utils)
  target_compile_definitions(ConvolutionHost PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_MODAL_LOOPS_PERMITTED=1)
  if(CH_BUILD_CLAP)
    include(cmake/ClapBusLayouts.cmake)
    clap_juce_extensions_plugin(TARGET ConvolutionHost CLAP_ID "com.lapowire.convolutionhost" CLAP_FEATURES audio-effect reverb)
    if(NOT TARGET ConvolutionHost_CLAP)
      message(FATAL_ERROR "Requested CLAP target was not created")
    endif()
  endif()
  foreach(format IN LISTS CH_JUCE_FORMATS)
    if(NOT TARGET ConvolutionHost_${format})
      message(FATAL_ERROR "Requested ${format} target was not created")
    endif()
  endforeach()
  if(BUILD_TESTING)
    add_executable(ch_plugin_tests tests/PluginTestMain.cpp tests/RealtimeGuard.cpp tests/ProcessorTests.cpp tests/MultichannelTests.cpp tests/EditorTests.cpp)
    target_include_directories(ch_plugin_tests PRIVATE src "${juce_SOURCE_DIR}/modules")
    target_compile_definitions(ch_plugin_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 $<$<CONFIG:Debug>:DEBUG=1;_DEBUG=1>)
    target_link_libraries(ch_plugin_tests PRIVATE ConvolutionHost)
    add_test(NAME plugin COMMAND ch_plugin_tests)
  endif()
endif()

if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set_source_files_properties(tests/RealtimeGuard.cpp PROPERTIES COMPILE_DEFINITIONS CH_WRAP_ALLOCATIONS=1)
  foreach(test_target ch_tests ch_plugin_tests)
    if(TARGET ${test_target})
      target_link_options(${test_target} PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
    endif()
  endforeach()
endif()

if(BUILD_TESTING)
  find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cmake_options COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cmake_formats.py)
  add_test(NAME native_headers COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_artifacts.py)
  add_test(NAME au_validation_script COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_au_validation.py)
  add_test(NAME test_report COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_test_report.py)
  add_test(NAME staging COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_stage_build.py)
  if(CH_BUILD_PLUGIN)
    add_executable(ch_wrapper_smoke tests/WrapperSmoke.cpp)
    target_link_libraries(ch_wrapper_smoke PRIVATE juce::juce_audio_utils)
    target_compile_definitions(ch_wrapper_smoke PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 CH_SMOKE_CLAP=$<BOOL:${CH_BUILD_CLAP}>)
    if(CH_BUILD_CLAP)
      target_link_libraries(ch_wrapper_smoke PRIVATE clap-core)
      add_test(NAME wrapper_clap COMMAND ch_wrapper_smoke --format clap --plugin $<TARGET_FILE:ConvolutionHost_CLAP>)
    endif()
    if(CH_BUILD_VST3)
      add_test(NAME wrapper_vst3 COMMAND ch_wrapper_smoke --format vst3 --plugin "${CMAKE_CURRENT_BINARY_DIR}/ConvolutionHost_artefacts/$<CONFIG>/VST3/CONVOLUTION HOST.vst3")
    endif()
    add_test(NAME wrapper_missing_file COMMAND ch_wrapper_smoke --format vst3 --plugin "${CMAKE_CURRENT_BINARY_DIR}/nonexistent.vst3")
    set_tests_properties(wrapper_missing_file PROPERTIES WILL_FAIL TRUE)
  endif()
endif()

if(CH_BUILD_PLUGIN)
  include(cmake/BuildProvenance.cmake)
endif()

foreach(identity_target ConvolutionHost ch_tests ch_plugin_tests ch_wrapper_smoke)
  if(TARGET ${identity_target})
    target_include_directories(${identity_target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  endif()
endforeach()
if(TEST staging)
  set_tests_properties(staging test_report PROPERTIES ENVIRONMENT "CH_STAGE_TEST_ARTIFACTS=${CMAKE_CURRENT_BINARY_DIR}/ConvolutionHost_artefacts/$<CONFIG>")
endif()
